cmake_minimum_required(VERSION 3.16)
project(waqti_runtime LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(waqti_gguf STATIC gguf_reader.cpp gguf_file.cpp q4_k_dequant.cpp tensor_decode.cpp qwen_reference.cpp attention_reference.cpp transformer_layer.cpp sequence_reference.cpp generation_reference.cpp tokenizer.cpp tensor_primitives.cpp toy_dense_model.cpp)
target_include_directories(waqti_gguf PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_compile_options(waqti_gguf PRIVATE -Wall -Wextra -Wpedantic)
if(ANDROID)
    add_library(waqti_runtime SHARED native_bridge.cpp)
    target_link_libraries(waqti_runtime PRIVATE waqti_gguf)
else()
    add_executable(gguf_reader_tests tests/gguf_reader_tests.cpp)
    target_link_libraries(gguf_reader_tests PRIVATE waqti_gguf)
    add_executable(runtime_primitives_tests tests/runtime_primitives_tests.cpp)
    target_link_libraries(runtime_primitives_tests PRIVATE waqti_gguf)
    add_executable(toy_dense_model_tests tests/toy_dense_model_tests.cpp)
    target_link_libraries(toy_dense_model_tests PRIVATE waqti_gguf)
    add_executable(matmul_benchmark tests/matmul_benchmark.cpp)
    target_link_libraries(matmul_benchmark PRIVATE waqti_gguf)
    add_executable(q4_k_dequant_tests tests/q4_k_dequant_tests.cpp)
    target_link_libraries(q4_k_dequant_tests PRIVATE waqti_gguf)
    add_executable(tensor_decode_tests tests/tensor_decode_tests.cpp)
    target_link_libraries(tensor_decode_tests PRIVATE waqti_gguf)
    add_executable(qwen_reference_tests tests/qwen_reference_tests.cpp)
    target_link_libraries(qwen_reference_tests PRIVATE waqti_gguf)
    add_executable(attention_reference_tests tests/attention_reference_tests.cpp)
    target_link_libraries(attention_reference_tests PRIVATE waqti_gguf)
    add_executable(sequence_reference_tests tests/sequence_reference_tests.cpp)
    target_link_libraries(sequence_reference_tests PRIVATE waqti_gguf)
    enable_testing()
    add_test(NAME gguf_reader_tests COMMAND gguf_reader_tests)
    add_test(NAME runtime_primitives_tests COMMAND runtime_primitives_tests)
    add_test(NAME toy_dense_model_tests COMMAND toy_dense_model_tests)
    add_test(NAME q4_k_dequant_tests COMMAND q4_k_dequant_tests)
    add_test(NAME tensor_decode_tests COMMAND tensor_decode_tests)
    add_test(NAME qwen_reference_tests COMMAND qwen_reference_tests)
    add_test(NAME attention_reference_tests COMMAND attention_reference_tests)
    add_test(NAME sequence_reference_tests COMMAND sequence_reference_tests)
endif()
